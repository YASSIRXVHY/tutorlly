   622→   622→export function tutorById(id: string | null | undefined): Tutor | undefined {
   623→   623→  if (!id) return undefined;
   624→   624→  return TUTORS.find((t) => t.id === id);
   625→   625→}

   627→   627→/** Resolved subjects a tutor teaches, in order. */
   628→   628→export function tutorSubjects(t: Tutor) {
   629→   629→  return t.teaches
   630→   630→    .map((te) => subjectById(te.subject))
   631→   631→    .filter((s): s is NonNullable<ReturnType<typeof subjectById>> => Boolean(s));
   632→   632→}

   635→   635→export function primarySubject(t: Tutor) {
   636→   636→  return tutorSubjects(t)[0];

   639→   639→/** Does this tutor teach any human language? */
   640→   640→export function isLanguageTutor(t: Tutor): boolean {
   641→   641→  return tutorSubjects(t).some((s) => s.cat === "languages");
   642→   642→}

   644→   644→/** The language a tutor teaches AND speaks natively, if any — drives the "native" filter. */
   645→   645→export function nativeTaughtLanguage(t: Tutor) {
   646→   646→  const taught = tutorSubjects(t).filter((s) => s.cat === "languages");
   647→   647→  return taught.find((s) =>
   648→   648→    t.spoken.some((sp) => subjectCodeMatches(sp.code, s.id) && sp.level === "Native")
   649→   649→  );
   650→   650→}

   652→   652→export function subjectCodeMatches(code: string, subjectId: string): boolean {
   653→   653→  // language codes vs subject ids: es↔spanish, zh↔mandarin, etc.
   654→   654→  const map: Record<string, string> = {
   655→   655→    en: "english",
   656→   656→    es: "spanish",
   657→   657→    fr: "french",
   658→   658→    de: "german",
   659→   659→    it: "italian",
   660→   660→    pt: "portuguese",
   661→   661→    ja: "japanese",
   662→   662→    ko: "korean",
   663→   663→    zh: "mandarin",
   664→   664→    ar: "arabic",
   665→   665→    ru: "russian",
   666→   666→  };
   667→   667→  return map[code] === subjectId;
   668→   668→}

   670→   670→/** Exam names this tutor coaches toward. */
   671→   671→export function tutorExams(t: Tutor): string[] {
   672→   672→  const out: string[] = [];
   673→   673→  for (const te of t.teaches) {
   674→   674→    if (subjectById(te.subject)?.cat === "test-prep") {
   675→   675→      out.push(subjectById(te.subject)!.label);
   676→   676→    } else if (subjectById(te.subject)?.cat === "languages") {
   677→   677→      out.push(...(LANG_EXAMS[te.subject] ?? []));
   678→   678→    }
   679→   679→  }
   680→   680→  return out;
   681→   681→}

   704→   704→export function initialsOf(name: string): string {
   705→   705→  return name
   706→   706→    .split(/\s+/)
   707→   707→    .slice(0, 2)
   708→   708→    .map((p) => p[0])
   709→   709→    .join("")
   710→   710→    .toUpperCase();
   711→   711→}
